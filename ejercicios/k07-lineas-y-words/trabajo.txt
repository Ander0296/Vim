public class Carrito {
    private final List<String> items = new ArrayList<>();
    private double total;

    public void agregar(String item, double precio) {
        items.add(item);
        total = total + precio;
//        System.out.println("agregado: " + item);
    }

    public double totalConIva() {
        return total * 1.15;
    }

    public String resumen() {
        return "Items: " + items.size() + " - Monto: " + total;
    }
}
