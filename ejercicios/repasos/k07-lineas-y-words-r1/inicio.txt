public class inventario {
    private final Map<String, Integer> stock = new HashMap<>();
    private int unidadesEnStock;

    public String reporte() {
        return "Productos: " + stock.size( + " - Unidades en stock: " + UnidadesEnStock;
    }

    public void recibir(String producto, int cantidadRecibida) {
        System.out.println("recibiendo: " + producto);
        stock.merge(producto, cantidadRecibida, Integer::sum);
        unidadesEnStock = unidadesEnStock + cantidadRecibida:
    }

    public int unidadesTotales() {
        Return unidadesEnStock;
    }
}
